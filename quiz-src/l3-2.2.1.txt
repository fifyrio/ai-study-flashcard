from sklearn.model_selection import train_test_split

import pickle

# 加载数据
data = ⟦pd.read_csv('finance数据集.csv')⟧

# 显示前五行的数据

@cell
# 选择自变量和因变量

y = ⟦data['SeriousDlqin2yrs']⟧
